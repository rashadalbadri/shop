    function loadState() {
      try {
        const raw = localStorage.getItem(STORAGE_KEY);
        if (!raw) return structuredClone(DEFAULT_STATE);
        const parsed = JSON.parse(raw);
        return {
          config: { ...DEFAULT_STATE.config, ...(parsed.config || {}) },
          invoices: Array.isArray(parsed.invoices) ? parsed.invoices : []
        };
      } catch (e) {
        return structuredClone(DEFAULT_STATE);
      }
    }

    function persistState(syncCloud = true) {
      localStorage.setItem(STORAGE_KEY, JSON.stringify(state));
      if (syncCloud && state.config.firebaseUrl) {
        pushToFirebase();
      }
    }

    async function pushToFirebase() {
      const base = state.config.firebaseUrl.trim().replace(/\/+$/, '');
      if (!base) return;
      try {
        await fetch(`${base}/queue_system.json`, {
          method: 'PUT',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify(state)
        });
        document.getElementById('syncStatusText').textContent = '(متصل بالسحابة)';
      } catch (err) {
        document.getElementById('syncStatusText').textContent = '(تعذر المزامنة)';
      }
    }

    async function pullFromFirebase() {
      const base = state.config.firebaseUrl.trim().replace(/\/+$/, '');
      if (!base) return;
      try {
        const res = await fetch(`${base}/queue_system.json`);
        if (!res.ok) return;
        const remote = await res.json();
        if (remote && remote.config) {
          state = {
            config: { ...DEFAULT_STATE.config, ...remote.config },
            invoices: Array.isArray(remote.invoices) ? remote.invoices : []
          };
          localStorage.setItem(STORAGE_KEY, JSON.stringify(state));
          document.getElementById('syncStatusText').textContent = '(متصل بالسحابة)';
          applyBranding();
          refreshAllViews();
        }
      } catch (e) {}
    }

    function createInvoice() {

      const custName = document.getElementById('custNameInput').value.trim() || 'زبون';
      const custPhone = document.getElementById('custPhoneInput').value.trim();
      const price = parseFloat(document.getElementById('invPriceInput').value) || 0;
      const workerShare = parseFloat(document.getElementById('invWorkerShareInput').value) || 0;

      if (!num || num <= 0) {
        showMsg('cashierMsg', 'يرجى إدخال رقم فاتورة صحيح.', false);
        return;
      }
      if (state.invoices.some((i) => Number(i.number) === num)) {

    function sendWhatsApp(mode) {
      const inv = state.invoices.find((i) => i.id === selectedTicketId);

      const phone = (inv.customerPhone || '').replace(/\D/g, '');

    async function startScanner() {
      if (qrScanner) return;
      const readerEl = document.getElementById('qrReader');
      qrScanner = new Html5Qrcode('qrReader');
      try {
        await qrScanner.start(
          { facingMode: 'environment' },
          { fps: 10, qrbox: { width: 210, height: 210 } },
          (decodedText) => {
            const parts = decodedText.split('|');
            if (parts[0] === 'INV' && parts[1] && parts[2]) {
              document.getElementById('claimInvNum').value = parts[1];
              document.getElementById('claimPinCode').value = parts[2];
              claimInvoiceManual();
              stopScanner();
            }
          },
          () => {}
        );
      } catch (e) {
        showMsg('workerMsg', 'تعذر تشغيل الكاميرا. تأكد من السماح بالوصول للكاميرا عبر HTTPS.', false);
        qrScanner = null;
      }
    }

    async function stopScanner() {
      if (!qrScanner) return;
      try {
        await qrScanner.stop();
        await qrScanner.clear();
      } catch (e) {}
      qrScanner = null;
      document.getElementById('qrReader').innerHTML = '';
    }

    function claimInvoiceManual() {
      const wId = document.getElementById('activeWorkerSelect').value;
      const worker = state.config.workers.find((w) => w.id === wId);
      const num = parseInt(document.getElementById('claimInvNum').value, 10);
      const code = document.getElementById('claimPinCode').value.trim();

      if (!worker) {
        showMsg('workerMsg', 'اختر الاسم أولاً.', false);
        return;
      }
      const inv = state.invoices.find((i) => Number(i.number) === num);
      if (!inv) {
        showMsg('workerMsg', `الفاتورة رقم (${num}) غير موجودة.`, false);
        return;
      }
      if (String(inv.code) !== code) {
        showMsg('workerMsg', 'كود التفعيل السري غير مطابق للفاتورة!', false);
        return;
      }
      if (inv.status === 'claimed') {
        showMsg('workerMsg', `هذه الفاتورة مسجلة مسبقاً باسم (${inv.workerName}) ولا يمكن تكرارها.`, false);
        return;
      }

      inv.status = 'claimed';
      inv.workerId = worker.id;
      inv.workerName = worker.name;
      persistState();

      document.getElementById('claimInvNum').value = '';
      document.getElementById('claimPinCode').value = '';
      showMsg('workerMsg', `تم تثبيت الفاتورة #${inv.number} باسم (${worker.name}) بنجاح.`);
      refreshAllViews();
    }

      `).join('');
    }

    function addWorker() {
      const name = document.getElementById('newWorkerName').value.trim();
      const pin = document.getElementById('newWorkerPin').value.trim();
      if (!name || !pin) return;
      state.config.workers.push({ id: 'w_' + Date.now(), name, pin });
      document.getElementById('newWorkerName').value = '';
      document.getElementById('newWorkerPin').value = '';
      persistState(false);
      renderWorkersConfigTable();
      populateWorkerSelects();
    }

    function updateWorkerField(id, field, val) {
      const w = state.config.workers.find((x) => x.id === id);
      if (w) {
        w[field] = val.trim();
        persistState(false);
        populateWorkerSelects();
      }
    }

    function removeWorker(id) {
      state.config.workers = state.config.workers.filter((x) => x.id !== id);
      persistState(false);
      renderWorkersConfigTable();
      populateWorkerSelects();
    }

    function setThemeColor(hex) {
      document.getElementById('cfgColorPicker').value = hex;
      state.config.themeColor = hex;
      applyBranding();
    }

    function handleLogoUpload(e) {
      const file = e.target.files?.[0];
      if (!file) return;
      const reader = new FileReader();
      reader.onload = () => {
        state.config.logoDataUrl = reader.result;
        applyBranding();
      };
      reader.readAsDataURL(file);
    }

    function applyPreset(type) {
      if (type === 'turban') {
        state.config.shopName = 'التمصير الملكي';
        state.config.workerTitle = 'المُمصّر';
        state.config.serviceTitle = 'تمصيرة';
        state.config.icon = '👑';
        state.config.themeColor = '#d4a638';
        state.config.defaultPrice = 3;
        state.config.defaultWorkerShare = 2;
      } else if (type === 'barber') {
        state.config.shopName = 'صالون النخبة للحلاقة';
        state.config.workerTitle = 'الحلاق';
        state.config.serviceTitle = 'حلاقة';
        state.config.icon = '✂️';
        state.config.themeColor = '#3b82f6';
        state.config.defaultPrice = 4;
        state.config.defaultWorkerShare = 2;
      } else if (type === 'wash') {
        state.config.shopName = 'مغسلة السريع للسيارات';
        state.config.workerTitle = 'العامل';
        state.config.serviceTitle = 'غسلة';
        state.config.icon = '🚗';
        state.config.themeColor = '#10b981';
        state.config.defaultPrice = 5;
        state.config.defaultWorkerShare = 2.5;
      }
      persistState(false);
      fillSettingsForm();
      applyBranding();
      showMsg('settingsMsg', 'تم تطبيق القالب بنجاح.');
    }

    function saveSettings() {
      state.config.shopName = document.getElementById('cfgShopName').value.trim() || 'المحل';
      state.config.tagline = document.getElementById('cfgTagline').value.trim();
      state.config.workerTitle = document.getElementById('cfgWorkerTitle').value.trim() || 'العامل';
      state.config.serviceTitle = document.getElementById('cfgServiceTitle').value.trim() || 'خدمة';
      state.config.icon = document.getElementById('cfgIcon').value.trim() || '👑';
      state.config.themeColor = document.getElementById('cfgColorPicker').value || '#d4a638';
      state.config.defaultPrice = parseFloat(document.getElementById('cfgDefaultPrice').value) || 0;
      state.config.defaultWorkerShare = parseFloat(document.getElementById('cfgDefaultWorkerShare').value) || 0;
      state.config.cashierPin = document.getElementById('cfgCashierPin').value.trim() || '1234';
      state.config.firebaseUrl = document.getElementById('cfgFirebaseUrl').value.trim();

      persistState();
      applyBranding();
      refreshAllViews();
      showMsg('settingsMsg', 'تم حفظ جميع إعدادات المحل بنجاح.');
    }

    function exportConfigFile() {
      const blob = new Blob([JSON.stringify(state.config, null, 2)], { type: 'application/json' });
      const url = URL.createObjectURL(blob);
      const a = document.createElement('a');
      a.href = url;
      a.download = 'shop-config.json';
      a.click();
      URL.revokeObjectURL(url);
    }

    function importConfigFile(e) {
      const file = e.target.files?.[0];
      if (!file) return;
      const reader = new FileReader();
      reader.onload = () => {
        try {
          const imported = JSON.parse(reader.result);
          state.config = { ...state.config, ...imported };
          persistState();
          fillSettingsForm();
          applyBranding();
          refreshAllViews();
          showMsg('settingsMsg', 'تم استيراد ملف الإعدادات بنجاح.');
        } catch (err) {
          showMsg('settingsMsg', 'ملف الإعدادات غير صالح.', false);
        }
      };
      reader.readAsText(file);
    }

    function refreshAllViews() {
      renderWorkerView();
      renderAccountingView();
    }

    applyBranding();
  </script>
